" ==============================================================================
" 2. GLOBAL SETTINGS & INTERFACE
" ==============================================================================
set background=dark
colorscheme industry
syntax on

set number                  " Abbreviated from 'nu'
set encoding=utf-8
set incsearch
set hlsearch

" ==============================================================================
" 3. INDENTATION & TABS (PEP 8 COMPLIANT)
" ==============================================================================
" General defaults matching your Python requirements
set tabstop=4
set softtabstop=4
set shiftwidth=4
set textwidth=79
set expandtab
set autoindent
set fileformat=unix

" Filetype specific overrides
augroup FileTypeSettings
  autocmd!
  " Web assets use 2 spaces instead of 4
  autocmd BufNewFile,BufRead *.js,*.html,*.css setlocal tabstop=2 softtabstop=2 shiftwidth=2
  
  " Flagging trailing whitespace dynamically per PEP 8
  autocmd BufRead,BufNewFile *.py,*.pyw,*.c,*.h match BadWhitespace /\s\+$/
augroup END

highlight BadWhitespace ctermbg=red guibg=darkred

" Code Folding Setup
set foldmethod=indent
set foldlevel=99

" ==============================================================================
" 4. KEY MAPS & SHORTCUTS
" ==============================================================================
" Custom Leader (Defaults to '\') definitions

" Spacebar toggles folds
nnoremap <space> za

" Navigation & Spacing Shortcuts
nnoremap <kEnter> G
nnoremap <CR> o<Esc>
nnoremap <S-CR> O<Esc>

" Flash screen on paste 'p' to give visual landing cues
nnoremap p p:let my_bg = synIDattr(synIDtrans(hlID('Normal')), 'bg', 'gui')<CR>:hi Normal guibg=lightred<CR>:redraw!<CR>:sleep 200 m<CR>:execute 'hi Normal guibg=' . my_bg<CR>
